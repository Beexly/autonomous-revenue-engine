Signal Origin · selection tools

bookmarklets

Drag any button below to your bookmarks bar. Then select text on any page and click the bookmark. Each one runs entirely in your browser on the current selection — no network, no account, no tracking. Results appear in a plain alert. Same local heuristics as the paste pages here, not platform metrics.

1 · bait-check

bait-check ▸ drag me

Scores the first sentence of the selection with a minimal port of the bait-hook rules: stock openers, “here’s why” frames, lectures, listicles, thread announcements, CTAs, vague intensifiers, minus credit for a concrete verb + number. Alerts CLEAN / SOFT BAIT / HARD BAIT with risk /10 and flags.

readable source
(function(){var s=(window.getSelection?String(window.getSelection()):"").replace(/\s+/g," ").trim();if(!s){alert("bait-check: select some text first.");return;}var line=s.split(/(?<=[.!?])\s/)[0]||s;var r=0,f=[];function h(re,p,m){if(re.test(line)){r+=p;f.push(m);}}h(/^(unpopular opinion|hot take|hear me out|let'?s be honest|am i the only one|wait for it)\b/i,3,"stock opener");h(/^(the (real |actual )?reason|here'?s why|nobody (talks|tells)|what they don'?t)\b/i,3,"clickbait 'here's why' frame");h(/^(you (need to|should|must)|stop (doing|saying)|if you'?re not)\b/i,2,"second-person lecture");h(/^\d+\s+(ways|tips|habits|mistakes|signs|reasons|hacks)\b/i,3,"listicle count");h(/^(a )?thread\b/i,2,"thread announcement");h(/\b(like if|rt this|retweet|comment if|follow for|drop a)\b/i,4,"engagement CTA");h(/\b(insane|crazy|wild|mind-blowing|game-changing|secret)\b/i,2,"vague intensifier");var c=(/\b(killed|paid|refused|built|measured|failed|shipped|cut|deferred|chose)\b/i.test(line)?1:0)+(/\d/.test(line)?1:0);if(c===0&&line.length>18){r+=2;f.push("no decision verb and no number");}else if(c>=2){r=Math.max(0,r-2);f.push("credit: concrete verb + number");}if(r>10)r=10;var v=r<=2?"CLEAN":r<=5?"SOFT BAIT":"HARD BAIT";alert("bait-check (local heuristic)\n"+v+" - risk "+r+"/10\n\nfirst line: "+line.slice(0,120)+"\n\n"+(f.length?f.join("\n"):"no flags"));})();

2 · first-160

first-160 ▸ drag me

Copies the first 160 characters of the selection to your clipboard — the fold-ruler first-screen band, in the browser chrome. If the clipboard is blocked, it falls back to a prompt you can copy from.

readable source
(function(){var s=window.getSelection?String(window.getSelection()):"";if(!s.trim()){alert("first-160: select some text first.");return;}var b=s.replace(/\s+/g," ").trim().slice(0,160);function d(){alert("first-160: copied "+b.length+" chars.\n\n"+b);}if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(b).then(d,function(){prompt("first-160: clipboard blocked, copy manually:",b);});}else{prompt("first-160: copy manually:",b);}})();

3 · cta-strip

cta-strip ▸ drag me

Shows the selection with common residue removed — hashtags, thread/bait glyphs, and follow-me / like-and-RT / link-in-bio / dm-me lines (the cta-strip patterns). What survives in the alert is the actual post.

readable source
(function(){var s=window.getSelection?String(window.getSelection()):"";if(!s.trim()){alert("cta-strip: select some text first.");return;}var t=s.replace(/\r\n/g,"\n");t=t.replace(/#[A-Za-z0-9_]+/g," ");t=t.replace(/[\u{1F9F5}\u{1F525}\u{1F447}\u{1F517}➡️]/gu," ");var P=[/\bfollow\s+(me|for|along)\b[^.\n]*/gi,/\blike\s+(and|&)\s*(rt|retweet|share)\b[^.\n]*/gi,/\b(rt|retweet)\s+(this|if)\b[^.\n]*/gi,/\blink\s+in\s+bio\b[^.\n]*/gi,/\b(comment|drop)\s+(below|a|your)\b[^.\n]*/gi,/\bsubscribe\b[^.\n]*/gi,/\bdm\s+me\b[^.\n]*/gi,/\b(thread|more)\s+below\b[^.\n]*/gi,/\bclick\s+(the|here|this)\b[^.\n]*/gi];for(var i=0;i<P.length;i++)t=t.replace(P[i]," ");t=t.replace(/[ \t]+/g," ").replace(/\n{3,}/g,"\n\n").trim();alert("cta-strip (local heuristic)\n\nresidue:\n"+(t||"(nothing left)"));})();
